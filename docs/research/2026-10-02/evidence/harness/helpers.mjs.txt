export const cleanEnv=()=>Object.fromEntries(Object.entries(process.env).filter(([k])=>!k.startsWith('DIFFMIND_')&&!['GITHUB_TOKEN','GH_TOKEN'].includes(k)));

export class Observer {
  constructor(name){this.name=name;this.start=Date.now();this.rows=[];fs.mkdirSync(path.join(root,'evidence'),{recursive:true});}
  log(task,details){const row={at:new Date().toISOString(),elapsed_ms:Date.now()-this.start,task,...details};this.rows.push(row);fs.writeFileSync(path.join(root,'evidence',this.name+'.json'),JSON.stringify(this.rows,null,2));console.log(JSON.stringify(row));return row;}
  async screenshot(page,name){await page.screenshot({path:path.join(root,'evidence',name+'.png'),fullPage:true});this.log('screenshot',{file:name+'.png'});}
  watch(page){page.on('pageerror',e=>this.log('browser_exception',{message:e.message}));page.on('console',m=>{if(m.type()==='error')this.log('browser_console_error',{message:m.text()})});}
}
export class StdioClient {
  constructor(home,obs){this.home=home;this.obs=obs;this.next=1;this.pending=new Map();}
  async connect(){this.proc=spawn(path.join(root,'diffmind'),['agent'],{cwd:root,env:{...cleanEnv(),DIFFMIND_HOME:this.home},stdio:['pipe','pipe','pipe']});this.proc.stderr.pipe(fs.createWriteStream(path.join(root,'evidence','agent-stderr.log'),{flags:'a'}));readline.createInterface({input:this.proc.stdout}).on('line',line=>{let msg;try{msg=JSON.parse(line)}catch{return}const p=this.pending.get(msg.id);if(p){clearTimeout(p.timer);this.pending.delete(msg.id);msg.error?p.reject(new Error(JSON.stringify(msg.error))):p.resolve(msg.result)}});await this.rpc('initialize',{protocolVersion:'2025-11-25',capabilities:{},clientInfo:{name:'diffmind-live-usability-observer',version:'1'}});this.proc.stdin.write(JSON.stringify({jsonrpc:'2.0',method:'notifications/initialized'})+'\n');this.obs.log('mcp_connected',{transport:'stdio'});}
  rpc(method,params){const id=this.next++;return new Promise((resolve,reject)=>{const timer=setTimeout(()=>{this.pending.delete(id);reject(new Error('RPC timeout '+method))},60000);this.pending.set(id,{resolve,reject,timer});this.proc.stdin.write(JSON.stringify({jsonrpc:'2.0',id,method,params})+'\n')});}
  async call(name,args={}){const start=Date.now();const r=await this.rpc('tools/call',{name,arguments:args});let data=r.structuredContent;if(data===undefined){const t=r.content?.find(c=>c.type==='text')?.text;try{data=JSON.parse(t)}catch{data=t}}this.obs.log('mcp_call',{name,args,duration_ms:Date.now()-start,isError:!!r.isError,data});return {error:!!r.isError,data};}
  async op(operation,selectors={},body){const name=operation.startsWith('get_')||operation.startsWith('list_')?'inspect_workspace':'manage_workspace';const r=await this.call(name,{operation,selectors,...(body===undefined?{}:{body})});return r.data;}
  async disconnect(){const p=this.proc;p.stdin.end();await Promise.race([new Promise(r=>p.once('exit',r)),sleep(12000)]);if(p.exitCode===null){p.kill('SIGTERM');await sleep(200)}this.obs.log('mcp_disconnected',{exit_code:p.exitCode});}
}
export async function waitIngestion(client,pid){for(let i=0;i<120;i++){const r=await client.op('get_ingestion',{pid});if(r.data?.status!=='running')return r.data;await sleep(250)}throw new Error('ingestion timeout');}
export async function launchUI(home,obs,extraEnv={},flags=[]){
 fs.mkdirSync(home,{recursive:true,mode:0o700});
 const socket=net.createServer();await new Promise(r=>socket.listen(0,'127.0.0.1',r));
 const port=socket.address().port;await new Promise(r=>socket.close(r));
 const proc=spawn(path.join(root,'diffmind'),['ui','--no-spa-rebuild','--host','127.0.0.1','--port',String(port),...flags],{cwd:root,env:{...cleanEnv(),DIFFMIND_HOME:home,...extraEnv},stdio:['ignore','pipe','pipe']});
 const log=fs.createWriteStream(path.join(root,'evidence',obs.name+'-server.log'));proc.stdout.pipe(log);proc.stderr.pipe(log);
 const url='http://127.0.0.1:'+port;
 for(let i=0;i<80;i++){
  try{
   const r=await fetch(url+'/healthz');
   if(r.ok){
    obs.log('ui_server_started',{url});
    return {proc,url,stop:async()=>{proc.kill('SIGTERM');await Promise.race([new Promise(r=>proc.once('exit',r)),sleep(10000)]);obs.log('ui_server_stopped',{exit_code:proc.exitCode})}};
   }
  }catch{}
  await sleep(100);
 }
 proc.kill('SIGTERM');throw new Error('UI startup failed');
}
