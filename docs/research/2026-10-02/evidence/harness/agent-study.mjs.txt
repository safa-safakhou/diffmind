import fs from 'node:fs';
import path from 'node:path';
import {execFileSync} from 'node:child_process';
import {root,chromium,Observer,StdioClient,sleep,waitIngestion,cleanEnv} from './helpers.mjs';
const obs=new Observer('agent-observations');
const client=new StdioClient(path.join(root,'agent-home'),obs);
let browser;
try {
  await client.connect();
  const tools=await client.rpc('tools/list',{});obs.log('discovered_tools',{count:tools.tools.length,names:tools.tools.map(t=>t.name)});
  await client.call('list_projects');
  await client.call('get_graph_summary');
  const rt=(await client.call('agent_runtime',{action:'status'})).data;
  const create=await client.op('create_project',{}, {name:'Live Agent Demo Shop'});
  const pid=create.data.id;
  await client.op('import_repositories',{pid},{provider:'local',root:path.join(root,'demo-shop/repositories'),dry_run:true});
  await client.op('list_repositories',{pid});
  await client.op('start_ingestion',{pid},{import:{provider:'local',root:path.join(root,'demo-shop/repositories')},concurrency:2});
  const baseline=await waitIngestion(client,pid);
  obs.log('first_graph_ready',{pid,status:baseline.status,analyzed:baseline.analyzed,reused:baseline.reused,run_id:baseline.graph_run_id});
  await client.call('get_graph_summary',{project:pid});
  await client.call('get_dependencies',{project:pid,service:'gateway',direction:'outbound'});
  await client.call('get_service',{project:pid,service:'checkout',detail:'full'});
  browser=await chromium.launch({headless:true});
  const page=await browser.newPage({viewport:{width:1440,height:900}});obs.watch(page);
  await page.goto(rt.dashboard_url+'#/projects/'+pid);await page.locator('.service-compact').first().waitFor({timeout:30000});
  await obs.screenshot(page,'A-first-agent-graph');
  execFileSync('sh',['<source-root>/scripts/apply-demo-shop-change.sh',path.join(root,'demo-shop')],{env:cleanEnv()});
  obs.log('synthetic_source_changed',{change:'checkout contract v2 committed'});
  await sleep(5000);
  const unchanged=await client.op('get_ingestion',{pid});
  await client.call('get_contracts',{project:pid,service:'checkout'});
  obs.log('after_change_without_refresh',{same_run:unchanged.data.graph_run_id===baseline.graph_run_id,wait_ms:5000});
  await browser.close();browser=null;
  await client.disconnect();
  let oldAvailable=true;try{await fetch(rt.dashboard_url,{signal:AbortSignal.timeout(1500)})}catch{oldAvailable=false}
  obs.log('disconnected_backend_availability',{old_url_available:oldAvailable});
  await client.connect();
  const reconnected=(await client.call('agent_runtime',{action:'status'})).data;
  await client.call('list_services',{project:pid});
  const beforeRefresh=await client.op('get_ingestion',{pid});
  obs.log('reconnect_snapshot',{same_run:beforeRefresh.data.graph_run_id===baseline.graph_run_id,dashboard_url:reconnected.dashboard_url});
  await client.op('start_ingestion',{pid},{});
  const changed=await waitIngestion(client,pid);
  obs.log('explicit_refresh_complete',{status:changed.status,analyzed:changed.analyzed,reused:changed.reused,run_id:changed.graph_run_id});
  await client.call('diff_contracts',{project:pid,from:baseline.graph_run_id,to:changed.graph_run_id,service:'checkout'});
  await client.op('start_ingestion',{pid},{});
  const repeated=await waitIngestion(client,pid);obs.log('unchanged_repeat',{status:repeated.status,analyzed:repeated.analyzed,reused:repeated.reused});
  fs.writeFileSync(path.join(root,'agent-state.json'),JSON.stringify({pid,baseline:baseline.graph_run_id,changed:changed.graph_run_id,latest:repeated.graph_run_id},null,2));
}catch(e){obs.log('study_driver_failure',{message:e.stack});process.exitCode=1}
finally{if(browser)await browser.close();if(client.proc)await client.disconnect()}
