import fs from 'node:fs';
import path from 'node:path';
import {root,chromium,Observer,launchUI,sleep} from './helpers.mjs';
const obs=new Observer('explorer-continuation');const pid='live-explorer-demo-shop';let service,browser;
try{
 browser=await chromium.launch({headless:true});const page=await browser.newPage({viewport:{width:1440,height:900}});obs.watch(page);
 service=await launchUI(path.join(root,'explorer-home'),obs);await page.goto(service.url+'#/projects/'+pid);await page.getByText('Last ingestion failed',{exact:false}).waitFor();await page.locator('.ingestion-result summary').click().catch(()=>{});
 obs.log('ingestion_failure_ui',{body_text:await page.locator('body').innerText()});await obs.screenshot(page,'C-ingestion-path-failure');
 await service.stop();service=null;
 service=await launchUI(path.join(root,'explorer-home'),obs,{PATH:root+path.delimiter+process.env.PATH});obs.log('temporary_process_environment_recovery',{change:'Added study binary directory to this UI process PATH only'});
 await page.goto(service.url+'#/projects/'+pid);await page.getByRole('button',{name:'Update graph',exact:true}).waitFor();const start=Date.now();await page.getByRole('button',{name:'Update graph',exact:true}).click();await page.locator('.service-compact').first().waitFor({timeout:30000});obs.log('ui_graph_ready',{automation_ms:Date.now()-start,service_nodes:await page.locator('.service-compact').count(),footer:await page.locator('footer').innerText()});await obs.screenshot(page,'C-initial-fit-1440');
 const labelMetrics=()=>page.locator('svg text').evaluateAll(nodes=>nodes.filter(n=>n.textContent==='gateway').map(n=>({text:n.textContent,font_size:parseFloat(getComputedStyle(n).fontSize),scale:n.getScreenCTM().a,screen_height:n.getBoundingClientRect().height})));
 obs.log('initial_label_readability',{labels:await labelMetrics()});const query=page.getByPlaceholder('Search service');await query.fill('gateway');await query.press('Enter');await sleep(700);obs.log('search_focus',{labels:await labelMetrics()});await obs.screenshot(page,'C-gateway-search-focus');
 await page.locator('.service-compact').filter({hasText:'gateway'}).click();await sleep(600);obs.log('gateway_mouse_selection',{body_text:(await page.locator('body').innerText()).slice(-9000)});await obs.screenshot(page,'C-gateway-evidence');
 const keyboard=await page.locator('.service-compact').filter({hasText:'checkout'}).evaluate(n=>({tabindex:n.getAttribute('tabindex'),role:n.getAttribute('role')}));obs.log('service_keyboard_attributes',{service:'checkout',...keyboard});obs.log('keyboard_connection_rows',{count:await page.locator('.connection-summary-row[tabindex="0"]').count()});
 await page.getByRole('button',{name:'PR impact',exact:true}).click();await sleep(500);obs.log('pr_local_repos_screen',{body_text:await page.locator('body').innerText()});await obs.screenshot(page,'C-pr-local-repositories');
 await page.goto(service.url+'#/projects/'+pid+'/compare');await page.getByRole('heading',{name:'Compare graph snapshots'}).waitFor();await sleep(200);obs.log('single_snapshot_compare_entry',{body_text:await page.locator('body').innerText()});await obs.screenshot(page,'C-single-snapshot-compare');
 await page.setViewportSize({width:1280,height:720});await page.goto(service.url+'#/projects/'+pid);await page.locator('.service-compact').first().waitFor();await obs.screenshot(page,'C-fit-1280');
 await page.setViewportSize({width:1920,height:1080});await page.reload();await page.locator('.service-compact').first().waitFor();await obs.screenshot(page,'C-fit-1920');
 fs.writeFileSync(path.join(root,'explorer-state.json'),JSON.stringify({pid},null,2));
}catch(e){obs.log('study_driver_failure',{message:e.stack});process.exitCode=1}
finally{if(browser)await browser.close();if(service)await service.stop()}
