import fs from 'node:fs';
import path from 'node:path';
import {root,chromium,Observer,launchUI,sleep} from './helpers.mjs';
const obs=new Observer('explorer-observations');let service,browser;
try{
 service=await launchUI(path.join(root,'explorer-home'),obs);
 browser=await chromium.launch({headless:true});
 const page=await browser.newPage({viewport:{width:1440,height:900}});obs.watch(page);
 await page.goto(service.url);await page.locator('.modal').waitFor();
 obs.log('first_use_modal',{text:await page.locator('.modal').innerText(),named_name_fields:await page.getByRole('textbox',{name:'Name',exact:true}).count(),dialog_role_count:await page.getByRole('dialog').count(),initial_focus:await page.evaluate(()=>document.activeElement.tagName)});
 await obs.screenshot(page,'C-empty-first-use');
 await page.keyboard.press('Escape');obs.log('escape_first_modal',{still_open:await page.locator('.modal').count()});
 const focus=[];for(let i=0;i<9;i++){await page.keyboard.press('Tab');focus.push(await page.evaluate(()=>({tag:document.activeElement.tagName,text:(document.activeElement.textContent||'').trim().slice(0,40),in_modal:!!document.activeElement.closest('.modal')})))}obs.log('modal_tab_sequence',{focus});
 await page.locator('.modal input').first().fill('Live Explorer Demo Shop');await page.getByRole('button',{name:'Create project',exact:true}).click();await page.locator('.modal').waitFor({state:'hidden'});
 const pid=decodeURIComponent(page.url().split('/projects/')[1]);obs.log('project_created_via_ui',{pid});
 await page.getByRole('button',{name:'Import & build',exact:true}).click();await page.getByRole('button',{name:'Local directory',exact:true}).click();
 const rootField=()=>page.locator('.modal .field').filter({has:page.locator('label',{hasText:/^Root directory$/})}).locator('input');
 await rootField().fill(path.join(root,'demo-shop/repositories'));await page.getByRole('checkbox',{name:'Dry run only'}).check();
 const previewResponse=page.waitForResponse(r=>r.request().method()==='POST'&&r.url().includes('import'));
 await page.getByRole('button',{name:'Preview import',exact:true}).click();const preview=await previewResponse;const result=await preview.json();await page.locator('.modal').waitFor({state:'hidden'});await sleep(200);
 obs.log('preview_ui_result',{status:preview.status(),count:result.count,candidate_names:(result.repos||[]).map(r=>r.name),body_text:await page.locator('body').innerText()});
 await obs.screenshot(page,'C-after-dry-run-preview');
 await page.getByRole('button',{name:'Import & build',exact:true}).click();await page.getByRole('button',{name:'Local directory',exact:true}).click();obs.log('reopen_import_after_preview',{root_value:await rootField().inputValue(),dry_run_checked:await page.getByRole('checkbox',{name:'Dry run only'}).isChecked()});
 await rootField().fill(path.join(root,'does-not-exist'));await page.getByRole('checkbox',{name:'Dry run only'}).check();
 const badResponse=page.waitForResponse(r=>r.request().method()==='POST'&&r.url().includes('import'));await page.getByRole('button',{name:'Preview import',exact:true}).click();const bad=await badResponse;await sleep(200);obs.log('invalid_import_root',{status:bad.status(),modal_still_open:await page.locator('.modal').count(),errors:await page.locator('.banner.error').allTextContents(),error_inside_modal:await page.locator('.modal .banner.error').count()});await obs.screenshot(page,'C-invalid-root-error');
 await rootField().fill(path.join(root,'demo-shop/repositories'));await page.getByRole('checkbox',{name:'Dry run only'}).uncheck();const start=Date.now();await page.getByRole('button',{name:'Import and build graph',exact:true}).click();await page.locator('.modal').waitFor({state:'hidden'});await page.locator('.service-compact').first().waitFor({timeout:30000});obs.log('ui_graph_ready',{automation_ms:Date.now()-start,service_nodes:await page.locator('.service-compact').count(),footer:await page.locator('footer').innerText()});await obs.screenshot(page,'C-initial-fit-1440');
 const labelMetrics=()=>page.locator('svg text').evaluateAll(nodes=>nodes.filter(n=>n.textContent==='gateway').map(n=>({text:n.textContent,font_size:parseFloat(getComputedStyle(n).fontSize),scale:n.getScreenCTM().a,screen_height:n.getBoundingClientRect().height})));
 obs.log('initial_label_readability',{labels:await labelMetrics()});
 const query=page.getByPlaceholder('Search service');await query.fill('gateway');await query.press('Enter');await sleep(600);obs.log('search_focus',{labels:await labelMetrics(),selected_detail:await page.locator('.details-panel').count()});await obs.screenshot(page,'C-gateway-search-focus');
 await page.locator('.service-compact').filter({hasText:'gateway'}).click();await sleep(500);obs.log('gateway_mouse_selection',{body_text:(await page.locator('body').innerText()).slice(-9000)});await obs.screenshot(page,'C-gateway-evidence');
 const keyboard=await page.locator('.service-compact').filter({hasText:'checkout'}).evaluate(n=>({tabindex:n.getAttribute('tabindex'),role:n.getAttribute('role')}));obs.log('service_keyboard_attributes',{service:'checkout',...keyboard});
 const connectionRows=await page.locator('.connection-summary-row[tabindex="0"]').count();obs.log('keyboard_connection_rows',{count:connectionRows});
 await page.getByRole('button',{name:'PR impact',exact:true}).click();await page.getByRole('heading',{name:'Pull request impact',exact:false}).waitFor({timeout:5000}).catch(()=>{});obs.log('pr_local_repos_screen',{body_text:await page.locator('body').innerText()});await obs.screenshot(page,'C-pr-local-repositories');
 fs.writeFileSync(path.join(root,'explorer-state.json'),JSON.stringify({pid,home:path.join(root,'explorer-home')},null,2));
 const agent=JSON.parse(fs.readFileSync(path.join(root,'agent-state.json')));await page.goto(service.url+'#/projects/'+pid+'/compare');obs.log('single_snapshot_compare_entry',{body_text:await page.locator('body').innerText()});
 await page.setViewportSize({width:1280,height:720});await page.goto(service.url+'#/projects/'+pid);await page.locator('.service-compact').first().waitFor();await obs.screenshot(page,'C-fit-1280');
 await page.setViewportSize({width:1920,height:1080});await page.reload();await page.locator('.service-compact').first().waitFor();await obs.screenshot(page,'C-fit-1920');
}catch(e){obs.log('study_driver_failure',{message:e.stack});process.exitCode=1}
finally{if(browser)await browser.close();if(service)await service.stop()}
