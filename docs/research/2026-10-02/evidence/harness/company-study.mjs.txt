import fs from 'node:fs';
import path from 'node:path';
import {randomBytes} from 'node:crypto';
import {root,chromium,Observer,launchUI,sleep} from './helpers.mjs';

 const secret=randomBytes(24).toString('hex');const adminSecret=randomBytes(24).toString('hex');

 await admin.locator('button:not(:disabled)').filter({hasText:/^Revoke$/}).first().click();await admin.getByRole('button',{name:'Confirm revocation',exact:true}).click();await admin.getByText('Token revoked.',{exact:false}).waitFor();const denied=await remote('tools/list',{});obs.log('revoked_agent_token',{http_status:denied.status});await obs.screenshot(admin,'B-token-revoked');
}catch(e){obs.log('study_driver_failure',{message:e.stack});process.exitCode=1}
finally{if(browser)await browser.close();if(service)await service.stop()}
