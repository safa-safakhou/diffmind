from pathlib import Path
import json,subprocess,os,sys,hashlib
root=Path("<TRIAL_ROOT>")
out=root/"scope-pair";out.mkdir(mode=0o700)
source=Path("<EARLIER_TRIAL_ROOT>/flask")
sys.path.insert(0,"<EARLIER_TRIAL_ROOT>/flask-runtime-packages")
import yaml
expected=json.loads((root/"source-corpus-verified/no-impact/oracle.json").read_text())["runtime_operations"]
(out/"oracle.json").write_text(json.dumps({"source_pin":"2c1b30d0503cfb064f1cb252e6614a06915a362a","expected_tutorial_operations":expected,"root_prefix":"examples/tutorial/","subdirectory_prefix":"","classification":"tutorial subset only; other root code unlabelled"},indent=2))
env={k:v for k,v in os.environ.items() if not k.startswith("DIFFMIND_") and k not in ("GITHUB_TOKEN","GH_TOKEN")}
env["DIFFMIND_HOME"]=str(out/"home")
rows=[]
for name,repo,prefix in [("enclosing",source,"examples/tutorial/"),("subdirectory",source/"examples/tutorial","")]:
 with (out/(name+".log")).open("w") as log:
  subprocess.run([str(root/"verified/diffmind"),"run","--repo",str(repo),"--out",str(out/name)],env=env,stdout=log,stderr=log,check=True,timeout=180)
 artifact=next((out/name).glob("*/diffmind.yaml"));facts=yaml.safe_load(artifact.read_text())
 evidence={e["id"]:e for e in facts["evidence"]}
 actual=set();pointers=[]
 for entry in facts["objects"]["http_endpoints"]:
  matching=[evidence[r] for r in entry["evidence_refs"] if evidence[r]["file"].startswith(prefix+"flaskr/")]
  if matching:
   details=entry["metadata"]["details"]
   actual.add((details["method"],details["path"]))
   pointers.extend({"operation":entry["name"],"file":e["file"],"line":e["start_line"]} for e in matching)
 expected_set={tuple(p) for p in expected}
 rows.append({"scope":name,"repository_path":facts["repository"]["path"],"expected":len(expected_set),"actual_tutorial_subset":len(actual),"missing":sorted(expected_set-actual),"unexpected":sorted(actual-expected_set),"source_pointers":pointers,"fact_sha256":hashlib.sha256(artifact.read_bytes()).hexdigest()})
(out/"results.json").write_text(json.dumps({"rows":rows},indent=2))
print(json.dumps([{k:v for k,v in r.items() if k!="source_pointers"} for r in rows],indent=2))
